using System;
using GestionRecherche.Data;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Logging;

namespace GestionRecherche.Business;

public class AlerteService : IAlerteService
{
    private readonly AppDbContext _dbContext;
    private readonly TimeProvider _timeProvider;
    private readonly ILogger<AlerteService> _logger;

    public AlerteService(
        AppDbContext dbContext,
        TimeProvider timeProvider,
        ILogger<AlerteService> logger)
    {
        _dbContext = dbContext;
        _timeProvider = timeProvider;
        _logger = logger;
    }

    public async Task<ResultatPassageAlerte> ExecuterPassageAsync(
    CancellationToken cancellationToken = default)
{
    // On enregistre l'heure exacte du début du passage.
    DateTime debutUtc =
        _timeProvider.GetUtcNow().UtcDateTime;

    _logger.LogInformation(
        "Début du passage du robot à {DebutUtc:o}.",
        debutUtc);

    // Valeurs utilisées pour construire le résultat,
    // même si une erreur se produit.
    int? delaiD1Jours = null;
    int? delaiD2Mois = null;
    int demandesExaminees = 0;

    // Au Jour 11, le robot ne crée pas encore d'anomalie.
     int anomaliesCreees = 0;

    try
    {
        /*
         * On relit D1 et D2 dans la base à chaque passage.
         *
         * AsNoTracking() :
         * nous lisons seulement les paramètres,
         * nous ne voulons pas les modifier.
         */
        List<ParametreSysteme> parametres =
            await _dbContext.ParametresSysteme
                .AsNoTracking()
                .Where(parametre =>
                    parametre.Cle == "D1" ||
                    parametre.Cle == "D2")
                .ToListAsync(cancellationToken);

        // On cherche séparément D1 et D2 dans la liste.
        ParametreSysteme? parametreD1 =
            parametres.SingleOrDefault(
                parametre => parametre.Cle == "D1");

        ParametreSysteme? parametreD2 =
            parametres.SingleOrDefault(
                parametre => parametre.Cle == "D2");

        /*
         * Si D1 ou D2 n'existe pas,
         * le robot ne peut pas continuer correctement.
         */
        if (parametreD1 is null || parametreD2 is null)
        {
            DateTime finUtc =
                _timeProvider.GetUtcNow().UtcDateTime;

            _logger.LogWarning(
                "Passage arrêté : D1 ou D2 est absent.");

            return new ResultatPassageAlerte(
                EstReussi: false,
                Code: "PARAMETRE_MANQUANT",
                Message:
                    "Le paramètre D1 ou D2 est absent.",
                DateDebutUtc: debutUtc,
                DateFinUtc: finUtc,
                DelaiD1Jours: null,
                DelaiD2Mois: null,
                DemandesExaminees: 0,
                AnomaliesCreees: 0);
        }

        /*
         * On vérifie les unités :
         * D1 doit être exprimé en jours.
         * D2 doit être exprimé en mois.
         */
        if (parametreD1.Unite != UniteParametre.Jours ||
            parametreD2.Unite != UniteParametre.Mois)
        {
            DateTime finUtc =
                _timeProvider.GetUtcNow().UtcDateTime;

            _logger.LogWarning(
                "Unités invalides : D1={UniteD1}, D2={UniteD2}.",
                parametreD1.Unite,
                parametreD2.Unite);

            return new ResultatPassageAlerte(
                EstReussi: false,
                Code: "UNITE_INVALIDE",
                Message:
                    "D1 doit être en jours et D2 en mois.",
                DateDebutUtc: debutUtc,
                DateFinUtc: finUtc,
                DelaiD1Jours: null,
                DelaiD2Mois: null,
                DemandesExaminees: 0,
                AnomaliesCreees: 0);
        }

        /*
         * Un délai égal à zéro ou négatif
         * n'est pas utilisable.
         */
        if (parametreD1.Valeur <= 0 ||
            parametreD2.Valeur <= 0)
        {
            DateTime finUtc =
                _timeProvider.GetUtcNow().UtcDateTime;

            _logger.LogWarning(
                "Valeurs invalides : D1={D1}, D2={D2}.",
                parametreD1.Valeur,
                parametreD2.Valeur);

            return new ResultatPassageAlerte(
                EstReussi: false,
                Code: "VALEUR_INVALIDE",
                Message:
                    "Les valeurs D1 et D2 doivent être positives.",
                DateDebutUtc: debutUtc,
                DateFinUtc: finUtc,
                DelaiD1Jours: null,
                DelaiD2Mois: null,
                DemandesExaminees: 0,
                AnomaliesCreees: 0);
        }

        // Les paramètres sont valides :
        // on récupère leurs valeurs.
        delaiD1Jours = parametreD1.Valeur;
        delaiD2Mois = parametreD2.Valeur;

        _logger.LogInformation(
            "Paramètres relus : D1={D1} jour(s), D2={D2} mois.",
            delaiD1Jours,
            delaiD2Mois);

        /*
         * On compte seulement les demandes :
         * 1. autorisées par le Conseil ;
         * 2. possédant une date d'autorisation.
         *
         * Au Jour 11, on les compte seulement.
         * La création des anomalies commencera au Jour 12.
         */
  /*
 * On recherche les demandes qui doivent être surveillées pour D1.
 *
 * Une demande nous intéresse uniquement si :
 *
 * 1. elle a été autorisée par le Conseil ;
 * 2. elle possède une date d'autorisation ;
 * 3. l'auteur n'a pas encore déclaré la soumission réelle.
 */
List<DemandePublication> demandesAutorisees =
    await _dbContext.DemandesPublication
        .Where(demande =>
            demande.StatutValidationInterne ==
                StatutValidationInterne.Autorisee
            &&
            demande.DateAutorisationConseilUtc != null
            &&
            demande.DateSoumissionReelleUtc == null)
        .ToListAsync(cancellationToken);


demandesExaminees = demandesAutorisees.Count;

/*
 * Date/heure actuelle du robot.
 *
 * On utilise TimeProvider au lieu de DateTime.UtcNow
 * parce que cela permet de tester facilement le service.
 */
DateTime maintenantUtc =
    _timeProvider.GetUtcNow().UtcDateTime;

/*
 * On examine chaque demande une par une.
 */
foreach (DemandePublication demande in demandesAutorisees)
{
    /*
     * Récupérer la date d'autorisation du Conseil.
     *
     * Nous savons qu'elle n'est pas null car
     * nous avons déjà testé DateAutorisationConseilUtc != null
     * dans la requête ci-dessus.
     */
    DateTime dateAutorisationUtc =
        demande.DateAutorisationConseilUtc!.Value;

    /*
     * Calculer la date limite D1.
     *
     * Exemple :
     * autorisation = 01/07/2026
     * D1 = 30 jours
     *
     * dateLimite = 31/07/2026
     */
    DateTime dateLimiteUtc =
        dateAutorisationUtc.AddDays(
            delaiD1Jours.Value);

    /*
     * Si nous sommes encore dans le délai,
     * aucune anomalie ne doit être créée.
     */
    if (maintenantUtc <= dateLimiteUtc)
    {
        continue;
    }

    /*
     * La date limite est dépassée.
     *
     * Nous devons maintenant vérifier
     * si une anomalie D1 ouverte existe déjà.
     *
     * Cela empêche le robot de créer
     * la même anomalie plusieurs fois.
     */
    bool anomalieDejaOuverte =
        await _dbContext.Anomalies
            .AnyAsync(
                anomalie =>
                    anomalie.DemandePublicationId ==
                        demande.Id
                    &&
                    anomalie.Type ==
                        TypeAnomalie.NonSoumission
                    &&
                    anomalie.Statut ==
                        StatutAnomalie.Ouverte,
                cancellationToken);

    /*
     * Si l'anomalie existe déjà,
     * nous ne faisons rien.
     */
    if (anomalieDejaOuverte)
    {
        continue;
    }

    /*
     * Calcul du nombre de jours de retard.
     */
    int joursDeRetard =
        Math.Max(
            1,
            (int)Math.Ceiling(
                (maintenantUtc - dateLimiteUtc).TotalDays));

    /*
     * Création d'un message compréhensible
     * pour l'administrateur / Chef de Bureau.
     */
    string message =
        $"La demande #{demande.Id} a été autorisée par le Conseil " +
        $"le {dateAutorisationUtc:dd/MM/yyyy HH:mm} UTC, " +
        $"mais aucune soumission réelle n'a été déclarée " +
        $"après le délai D1 de {delaiD1Jours.Value} jour(s). " +
        $"Retard : {joursDeRetard} jour(s).";


    /*
     * Création de l'objet Anomalie.
     */
    Anomalie anomalie =
        new Anomalie
        {
            DemandePublicationId = demande.Id,

            Type = TypeAnomalie.NonSoumission,

            Statut = StatutAnomalie.Ouverte,

            DateDetectionUtc = maintenantUtc,

            DureeJours = joursDeRetard,

            Message = message
        };


    /*
     * On prépare l'insertion dans la base.
     */
    _dbContext.Anomalies.Add(anomalie);

    /*
     * Compteur utilisé dans ResultatPassageAlerte.
     */
    anomaliesCreees++;

    _logger.LogWarning(
        "Anomalie D1 créée pour la demande {DemandeId}. " +
        "Retard={JoursDeRetard} jour(s).",
        demande.Id,
        joursDeRetard);
}


/*
 * ================================
 * JOUR 13 : DÉTECTION D2
 * ================================
 *
 * D2 concerne les articles qui :
 *
 * 1. ont déjà été soumis ;
 * 2. possèdent un statut éditeur ;
 * 3. ne sont pas terminés (Published / Rejected) ;
 * 4. possèdent une date de dernière mise à jour ;
 * 5. n'ont pas évolué depuis D2 mois.
 */
foreach (DemandePublication demande
    in demandesStagnantesPossibles)
{
    cancellationToken.ThrowIfCancellationRequested();

    DateTime derniereMiseAJourUtc =
        demande.DateDerniereMiseAJourUtc!.Value;

    /*
     * D2 est exprimé en mois.
     *
     * Exemple :
     *
     * dernière mise à jour = 01/02/2026
     * D2 = 6 mois
     *
     * limite = 01/08/2026
     */
    DateTime dateLimiteStagnationUtc =
        derniereMiseAJourUtc.AddMonths(
            delaiD2Mois!.Value);

    /*
     * L'article n'est pas encore stagnant.
     */
    if (maintenantUtc <= dateLimiteStagnationUtc)
    {
        continue;
    }

    /*
     * Vérifier si une anomalie Stagnation ouverte
     * existe déjà pour cette demande.
     */
    bool anomalieDejaOuverte =
        await _dbContext.Anomalies
            .AnyAsync(
                anomalie =>
                    anomalie.DemandePublicationId ==
                        demande.Id

                    &&

                    anomalie.Type ==
                        TypeAnomalie.Stagnation

                    &&

                    anomalie.Statut ==
                        StatutAnomalie.Ouverte,
                cancellationToken);


    /*
     * Si elle existe déjà,
     * on ne crée pas de doublon.
     */
    if (anomalieDejaOuverte)
    {
        continue;
    }

    /*
     * Calcul du nombre de jours depuis
     * la dernière mise à jour.
     */
    int joursSansMiseAJour =
        Math.Max(
            1,
            (int)Math.Floor(
                (maintenantUtc -
                 derniereMiseAJourUtc).TotalDays));

    /*
     * Message de l'anomalie.
     */
    string message =
        $"La demande #{demande.Id} est stagnante. " +
        $"Son dernier changement de statut date du " +
        $"{derniereMiseAJourUtc:dd/MM/yyyy HH:mm} UTC. " +
        $"Le délai D2 de {delaiD2Mois.Value} mois est dépassé. " +
        $"Durée sans mise à jour : " +
        $"{joursSansMiseAJour} jour(s). " +
        $"Statut actuel : " +
        $"{demande.StatutEditeurCourant}.";

    /*
     * Créer l'anomalie.
     */
    Anomalie anomalie =
        new Anomalie
        {
            DemandePublicationId =
                demande.Id,

            Type =
                TypeAnomalie.Stagnation,

            Statut =
                StatutAnomalie.Ouverte,

            DateDetectionUtc =
                maintenantUtc,

            DureeJours =
                joursSansMiseAJour,

            Message =
                message
        };

    _dbContext.Anomalies.Add(anomalie);

    anomaliesCreees++;

    _logger.LogWarning(
        "Anomalie D2 créée pour la demande {DemandeId}. " +
        "JoursSansMiseAJour={JoursSansMiseAJour}. " +
        "Statut={Statut}.",
        demande.Id,
        joursSansMiseAJour,
        demande.StatutEditeurCourant);
}

        demande.DateDerniereMiseAJourUtc!.Value;

    /*
     * D2 est exprimé en mois.
     *
     * Exemple :

         * Une annulation signifie généralement que
         * l'application est en train de s'arrêter.
         *
         * Ce n'est pas une erreur technique :
         * on laisse donc l'annulation continuer.
         */
        _logger.LogInformation(
            "Le passage du robot a été annulé.");

        throw;
    }
    catch (Exception exception)
    {
        /*
         * Ici, on capture les vraies erreurs techniques :
         * problème SQL, connexion impossible, etc.
         */
        DateTime finUtc =
            _timeProvider.GetUtcNow().UtcDateTime;

        _logger.LogError(
            exception,
            "Erreur technique pendant le passage du robot.");

        return new ResultatPassageAlerte(
            EstReussi: false,
            Code: "ERREUR_TECHNIQUE",
            Message:
                "Une erreur technique a empêché le passage du robot.",

private async Task<int> DetecterNonSoumissionsAsync(
    DateTime maintenantUtc,
    int delaiD1Jours,
    CancellationToken cancellationToken)
{
    DateTime dateSeuilAutorisationUtc =
        maintenantUtc.AddDays(-delaiD1Jours);

    List<DemandePublication> demandesEnRetard =
        await _dbContext.DemandesPublication
            .Where(demande =>
                demande.StatutValidationInterne ==
                    StatutValidationInterne.Autorisee &&

                demande.DateAutorisationConseilUtc != null &&

                demande.DateAutorisationConseilUtc <
                    dateSeuilAutorisationUtc &&

                demande.DateSoumissionReelleUtc == null)

            .Where(demande =>
                !demande.Anomalies.Any(anomalie =>
                    anomalie.Type ==
                        TypeAnomalie.NonSoumission &&

                    anomalie.Statut ==
                        StatutAnomalie.Ouverte))

            .ToListAsync(cancellationToken);

    foreach (DemandePublication demande in demandesEnRetard)
    {
        DateTime dateAutorisationUtc =
            demande.DateAutorisationConseilUtc!.Value;

        DateTime dateEcheanceUtc =
            dateAutorisationUtc.AddDays(delaiD1Jours);

        int joursRetard = Math.Max(
            1,
            (int)Math.Ceiling(
                (maintenantUtc - dateEcheanceUtc).TotalDays));

        var anomalie = new Anomalie
        {
            DemandePublicationId = demande.Id,

            Type = TypeAnomalie.NonSoumission,

            Statut = StatutAnomalie.Ouverte,

            DateDetectionUtc = maintenantUtc,

            DureeJours = joursRetard,

            Message =
                $"La demande n°{demande.Id} n'a pas été déclarée " +
                $"comme soumise. Le délai D1 de {delaiD1Jours} " +
                $"jour(s) est dépassé de {joursRetard} jour(s)."
        };

        _dbContext.Anomalies.Add(anomalie);
    }

    if (demandesEnRetard.Count > 0)
    {
        await _dbContext.SaveChangesAsync(
            cancellationToken);
    }

    return demandesEnRetard.Count;
}



private async Task<ResultatDetection> DetecterStagnationsAsync(
    DateTime maintenantUtc,
    int delaiD2Mois,
    CancellationToken cancellationToken)
{
    /*
     * On récupère seulement les demandes qui peuvent être
     * surveillées par D2.
     *
     * D1 n'est pas concerné ici :
     * DateSoumissionReelleUtc doit obligatoirement exister.
     */
    List<DemandePublication> demandesASurveiller =
        await _dbContext.DemandesPublication
            .AsNoTracking()
            .Where(demande =>
                demande.StatutValidationInterne ==
                    StatutValidationInterne.Autorisee &&

                demande.DateSoumissionReelleUtc != null &&

                demande.DateDerniereMiseAJourUtc != null &&

                demande.StatutEditeurCourant.HasValue &&

                demande.StatutEditeurCourant !=
                    StatutEditeur.Published &&

                demande.StatutEditeurCourant !=
                    StatutEditeur.Rejected)
            .ToListAsync(cancellationToken);

    /*
     * On charge en une seule requête les demandes qui
     * possèdent déjà une anomalie D2 ouverte.
     *
     * HashSet permet ensuite de vérifier très rapidement
     * si un identifiant est déjà présent.
     */
    HashSet<int> demandesAvecD2Ouverte =
        (await _dbContext.Anomalies
            .AsNoTracking()
            .Where(anomalie =>
                anomalie.Type == TypeAnomalie.Stagnation &&

                anomalie.Statut == StatutAnomalie.Ouverte)
            .Select(anomalie =>
                anomalie.DemandePublicationId)
            .ToListAsync(cancellationToken))
        .ToHashSet();

    int anomaliesCreees = 0;

    foreach (DemandePublication demande in demandesASurveiller)
    {
        cancellationToken.ThrowIfCancellationRequested();

        DateTime derniereMiseAJourUtc =
            demande.DateDerniereMiseAJourUtc!.Value;

        /*
         * On utilise AddMonths et non pas D2 × 30 jours.
         *
         * Six mois calendaires ne correspondent pas toujours
         * au même nombre de jours.
         */
        DateTime dateLimiteStagnationUtc =
            derniereMiseAJourUtc.AddMonths(delaiD2Mois);

        /*
         * Si maintenant est avant ou exactement à la limite,
         * le délai n'est pas encore dépassé.
         */
        if (maintenantUtc <= dateLimiteStagnationUtc ||
            demandesAvecD2Ouverte.Contains(demande.Id))
        {
            continue;
        }


        StatutEditeur statutActuel =
            demande.StatutEditeurCourant!.Value;

        /*
         * DureeJours représente ici le nombre total de jours
         * sans aucune mise à jour.
         */
        int joursSansMiseAJour = Math.Max(
            1,
            (int)Math.Floor(
                (maintenantUtc -
                 derniereMiseAJourUtc).TotalDays));

        var anomalie = new Anomalie
        {
            DemandePublicationId = demande.Id,

            Type = TypeAnomalie.Stagnation,

            Statut = StatutAnomalie.Ouverte,

            DateDetectionUtc = maintenantUtc,

            DureeJours = joursSansMiseAJour,

            Message =
                $"La demande #{demande.Id} est sans évolution " +
                $"depuis {joursSansMiseAJour} jour(s). " +
                $"Dernière mise à jour : " +
                $"{derniereMiseAJourUtc:dd/MM/yyyy HH:mm} UTC. " +
                $"Le délai D2 de {delaiD2Mois} mois est dépassé. " +
                $"Statut éditeur actuel : {statutActuel}."
        };

        _dbContext.Anomalies.Add(anomalie);

        /*
         * On ajoute aussi l'identifiant au HashSet.
         * Cela protège le passage courant.
         */
        demandesAvecD2Ouverte.Add(demande.Id);

        anomaliesCreees++;

        _logger.LogWarning(
            "Anomalie D2 créée. " +
            "Demande={DemandeId}, " +
            "Statut={Statut}, " +
            "Jours={Jours}.",
            demande.Id,
            statutActuel,
            joursSansMiseAJour);
    }

    return new ResultatDetection(
        DemandesExaminees: demandesASurveiller.Count,
        AnomaliesCreees: anomaliesCreees);
}
